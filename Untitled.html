import numpy as np
import matplotlib.pyplot as plt
from abc import ABC, abstractmethod
from sklearn.model_selection import train_test_split, LeaveOneOut
from sklearn.neighbors import KNeighborsRegressor
from sklearn.kernel_ridge import KernelRidge
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF
from sklearn.preprocessing import SplineTransformer
from sklearn.linear_model import LinearRegression
import statsmodels.api as sm
from scipy.interpolate import UnivariateSpline, CubicSpline
from numpy.polynomial.chebyshev import chebvander
from numpy.polynomial.legendre import legvander

# ==========================================
# 1. CORE INTERFACE & DATA PIPELINE
# ==========================================
class DataGenerator:
    """Handles the production and partitioning of the mathematical data stream."""
    def __init__(self, n_samples=150, noise_std=0.25, seed=42):
        self.n_samples = n_samples
        self.noise_std = noise_std
        self.seed = seed

    def generate(self):
        """Generates synthetic non-linear data avoiding array broadcasting bugs."""
        np.random.seed(self.seed)
        X = np.sort(6 * np.random.rand(self.n_samples, 1), axis=0)
        y_true = np.sin(X).ravel() + np.exp(-X**2).ravel()
        noise = np.random.normal(0, self.noise_std, size=self.n_samples) 
        y = y_true + noise
        return X, y


class BaseModel(ABC):
    """Abstract Base Class establishing structural interface for all smoothing tracks."""
    @abstractmethod
    def fit_and_predict(self, X_train, y_train, X_plot, X_plot_2d):
        """Must fit the algorithms and return a dict of {'Name': (color, y_pred)}."""
        pass


# ==========================================
# 2. INDIVIDUAL ALGORITHM CLASSES
# ==========================================
class KernelMethods(BaseModel):
    """Encapsulates 3 distinct local Kernel-based smoothing estimators."""
    def __init__(self, nw_bw=0.3, krr_alpha=0.05, gpr_alpha=0.1):
        self.nw_bw = nw_bw
        self.krr_alpha = krr_alpha
        self.gpr_alpha = gpr_alpha

    def fit_and_predict(self, X_train, y_train, X_plot, X_plot_2d):
        X_train_flat = X_train.ravel()
        X_plot_flat = X_plot_2d.ravel()
        
        nw = sm.nonparametric.KernelReg(endog=y_train, exog=X_train_flat, var_type='c', bw=[self.nw_bw])
        y_nw, _ = nw.fit(X_plot_flat)
        
        krr = KernelRidge(alpha=self.krr_alpha, kernel='rbf', gamma=1.0).fit(X_train, y_train)
        y_krr = krr.predict(X_plot_2d)
        
        gpr = GaussianProcessRegressor(kernel=RBF(length_scale=0.5), alpha=self.gpr_alpha).fit(X_train, y_train)
        y_gpr = gpr.predict(X_plot_2d)
        
        return {
            "Nadaraya-Watson": ("red", y_nw),
            "Kernel Ridge": ("blue", y_krr),
            "Gaussian Process": ("green", y_gpr)
        }


class SplineMethods(BaseModel):
    """Encapsulates Spline mathematical tracks optimized via Cross-Validation."""
    def __init__(self, n_knots=12, degree=3):
        self.n_knots = n_knots
        self.degree = degree

    def _optimize_smoothing_factor(self, X_sorted, y_sorted):
        """Finds the mathematically ideal value for 's' using Leave-One-Out CV."""
        s_candidates = np.logspace(-2, 1, 30)  # Grid search from 0.01 to 10.0
        loo = LeaveOneOut()
        best_s = 1.0
        min_press_score = float('inf')

        for s_val in s_candidates:
            press_score = 0.0
            try:
                # Iterate through cross-validation folds
                for train_idx, test_idx in loo.split(X_sorted):
                    X_cv_train, X_cv_test = X_sorted[train_idx], X_sorted[test_idx]
                    y_cv_train, y_cv_test = y_sorted[train_idx], y_sorted[test_idx]
                    
                    # Fit a local tracking spline on out-of-sample combinations
                    spline_cv = UnivariateSpline(X_cv_train, y_cv_train, s=s_val, k=3)
                    pred = spline_cv(X_cv_test)
                    press_score += (y_cv_test[0] - pred[0]) ** 2
                
                if press_score < min_press_score:
                    min_press_score = press_score
                    best_s = s_val
            except Exception:
                continue # Catch numerical boundary fails on extreme overfitting constraints

        print(f"[CV OPTIMIZER] Optimal Smoothing Factor Found: s = {best_s:.4f}")
        return best_s

    def fit_and_predict(self, X_train, y_train, X_plot, X_plot_2d):
        sort_idx = np.argsort(X_train.ravel())
        X_sorted = X_train.ravel()[sort_idx]
        y_sorted = y_train[sort_idx]

        # 1. Run optimization engine for Univariate Smoothing Splines
        optimal_s = self._optimize_smoothing_factor(X_sorted, y_sorted)

        # Algo 1: Cubic Spline (Exact Boundary Node Interpolation)
        cs = CubicSpline(X_sorted, y_sorted)
        y_cs = cs(X_plot.ravel())
        
        # Algo 2: Smoothing Spline using optimized parameter
        ss = UnivariateSpline(X_sorted, y_sorted, s=optimal_s)
        y_ss = ss(X_plot.ravel())
        
        # Algo 3: B-Spline Regression
        transformer = SplineTransformer(n_knots=self.n_knots, degree=self.degree)
        X_train_bs = transformer.fit_transform(X_train)
        bs_regressor = LinearRegression().fit(X_train_bs, y_train)
        y_bs = bs_regressor.predict(transformer.transform(X_plot_2d))
        
        return {
            "Cubic Spline (Interp)": ("orange", y_cs),
            f"Smoothing Spline (s={optimal_s:.2f})": ("purple", y_ss),
            "B-Spline Regression": ("brown", y_bs)
        }


class NearestNeighborsMethod(BaseModel):
    """Encapsulates geographic proximity clustering models."""
    def __init__(self, n_neighbors=10):
        self.n_neighbors = n_neighbors

    def fit_and_predict(self, X_train, y_train, X_plot, X_plot_2d):
        knn = KNeighborsRegressor(n_neighbors=self.n_neighbors).fit(X_train, y_train)
        return {f"k-NN (k={self.n_neighbors})": ("teal", knn.predict(X_plot_2d))}


class OrthogonalSeriesMethod(BaseModel):
    """Encapsulates global mathematical polynomial series decomposition tracks."""
    def __init__(self, degree=5):
        self.degree = degree

    def _scale_to_range(self, X_data, current_min=0.0, current_max=6.0):
        """Maps data linearly to [-1, 1] for Chebyshev/Legendre stability constraints."""
        return 2.0 * (X_data - current_min) / (current_max - current_min) - 1.0

    def fit_and_predict(self, X_train, y_train, X_plot, X_plot_2d):
        X_train_scaled = self._scale_to_range(X_train.ravel())
        X_plot_scaled = self._scale_to_range(X_plot)

        # Algo 1: Chebyshev Polynomial Series
        X_train_cheb = chebvander(X_train_scaled, self.degree)
        coefs_cheb, _, _, _ = np.linalg.lstsq(X_train_cheb, y_train, rcond=None)
        X_plot_cheb = chebvander(X_plot_scaled, self.degree)
        y_cheb = X_plot_cheb @ coefs_cheb

        # Algo 2: Legendre Polynomial Series
        X_train_leg = legvander(X_train_scaled, self.degree)
        coefs_leg, _, _, _ = np.linalg.lstsq(X_train_leg, y_train, rcond=None)
        X_plot_leg = legvander(X_plot_scaled, self.degree)
        y_leg = X_plot_leg @ coefs_leg

        return {
            "Chebyshev Series": ("crimson", y_cheb),
            "Legendre Series": ("darkviolet", y_leg)
        }


# ==========================================
# 3. EXECUTIVE ORCHESTRATION CONTROLLER
# ==========================================
class RegressionOrchestrator:
    """Executes data splitting, triggers separate model blocks, and prints evaluation plots."""
    def __init__(self, X, y, test_size=0.2):
        self.X = X
        self.y = y
        self.X_train, _, self.y_train, _ = train_test_split(X, y, test_size=test_size, random_state=42)
        self.X_plot = np.linspace(0, 6, 500)
        self.X_plot_2d = self.X_plot[:, np.newaxis]

    def _render_panel(self, ax, results_dict, title):
        """Standardized plotting method to draw individual subplots uniformly."""
        ax.scatter(self.X, self.y, color='gray', alpha=0.3, label='Data Samples')
        for name, (color, y_pred) in results_dict.items():
            linestyle = '--' if 'Interp' in name else '-'
            ax.plot(self.X_plot, y_pred, color=color, linewidth=2, linestyle=linestyle, label=name)
        ax.set_title(title)
        ax.legend()
        ax.grid(True, alpha=0.2)
        ax.set_ylim(-1.8, 1.8)  # Prevents any remaining erratic boundaries from scaling the axis out

    def run(self):
        kernel_runner = KernelMethods()
        spline_runner = SplineMethods()
        neighbor_runner = NearestNeighborsMethod()
        series_runner = OrthogonalSeriesMethod()

        kernel_res = kernel_runner.fit_and_predict(self.X_train, self.y_train, self.X_plot, self.X_plot_2d)
        spline_res = spline_runner.fit_and_predict(self.X_train, self.y_train, self.X_plot, self.X_plot_2d)
        neighbor_res = neighbor_runner.fit_and_predict(self.X_train, self.y_train, self.X_plot, self.X_plot_2d)
        series_res = series_runner.fit_and_predict(self.X_train, self.y_train, self.X_plot, self.X_plot_2d)

        fig, axs = plt.subplots(2, 2, figsize=(15, 11))
        self._render_panel(axs[0, 0], kernel_res, "Kernel Methods (3 Algorithms)")
        self._render_panel(axs[0, 1], spline_res, "Spline Methods (3 Algorithms)")
        self._render_panel(axs[1, 0], neighbor_res, "Nearest Neighbors")
        self._render_panel(axs[1, 1], series_res, "Orthogonal Polynomial Series")
        
        plt.tight_layout()
        plt.show()


# ==========================================
# 4. ENTRY EXECUTION POINT
# ==========================================
if __name__ == "__main__":
    generator = DataGenerator()
    X_raw, y_raw = generator.generate()
    
    orchestrator = RegressionOrchestrator(X_raw, y_raw)
    orchestrator.run()
